     3→/**
     4→ * CustomerPortal — the public, unauthenticated customer surface.
     5→ *
     6→ * Three tabs:
     7→ *   1. Submit a ticket   → POST /api/tickets
     8→ *   2. Check ticket status → GET /api/tickets/lookup?ref=…&token=…
     9→ *   3. Help me now        → POST /api/chat (AI chatbot)
    10→ *
    11→ * Security notes (mirrored on the API side):
    12→ *  - All customer input is rendered as plain text. React's default escaping
    13→ *    is our XSS defense — we never reach for dangerouslySetInnerHTML.
    14→ *  - The `lookupToken` returned by the submit endpoint is the customer's
    15→ *    only handle on their ticket. We display it once on submit and pre-fill
    16→ *    the lookup form via in-memory state when they click "View my ticket".
    17→ *    We never put it in a URL fragment the browser would persist or log.
    18→ *  - Rate-limit (429) responses render a friendly countdown.
    19→ */

   144→/**
   145→ * Rate-limit retry countdown.
   146→ *
   147→ * Implemented as a component (not a hook) so the state initializer runs on
   148→ * mount — no synchronous setState-in-effect, which keeps the linter happy
   149→ * and avoids a cascading render. The parent conditionally renders this
   150→ * (only when `rateLimitReset` is set), so each 429 mounts a fresh countdown
   151→ * with the right starting value.
   152→ */
   153→function RetryCountdown({
   154→  seconds,
   155→  onDone,
   156→}: {
   157→  seconds: number
   158→  onDone: () => void
   159→}) {
   160→  const [remaining, setRemaining] = useState(seconds)
   161→  // Hold onDone in a ref so the interval effect can stay mount-only.
   162→  const onDoneRef = useRef(onDone)
   163→  useEffect(() => {
   164→    onDoneRef.current = onDone
   165→  }, [onDone])
   166→
   167→  useEffect(() => {
   168→    if (seconds <= 0) {
   169→      onDoneRef.current()
   170→      return
   171→    }
   172→    const id = setInterval(() => {
   173→      setRemaining((r) => {
   174→        if (r <= 1) {
   175→          clearInterval(id)
   176→          onDoneRef.current()
   177→          return 0
   178→        }
   179→        return r - 1
   180→      })
   181→    }, 1000)
   182→    return () => clearInterval(id)
   183→  }, [seconds])
   184→
   185→  return (
   186→    <span className="font-semibold tabular-nums" aria-live="polite">
   187→      {remaining}s
   188→    </span>
   189→  )
   190→}

   196→const ticketSchema = z.object({
   197→  subject: z
   198→    .string()
   199→    .min(3, 'Subject must be at least 3 characters.')
   200→    .max(LIMITS.MAX_SUBJECT, `Subject must be under ${LIMITS.MAX_SUBJECT} characters.`),
   201→  body: z
   202→    .string()
   203→    .min(10, 'Please give us a bit more detail (at least 10 characters).')
   204→    .max(LIMITS.MAX_BODY, `Description must be under ${LIMITS.MAX_BODY} characters.`),
   205→  customerEmail: z
   206→    .string()
   207→    .min(1, 'Email is required.')
   208→    .email('That email address does not look right.'),
   209→  customerName: z
   210→    .string()
   211→    .max(LIMITS.MAX_NAME, `Name must be under ${LIMITS.MAX_NAME} characters.`)
   212→    .optional(),
   213→})
   214→
   215→type TicketFormValues = z.infer<typeof ticketSchema>

   217→type SubmitTabProps = {
   218→  onSubmitted: (result: SubmitResult) => void

   221→function SubmitTicketForm({ onSubmitted }: SubmitTabProps) {
   222→  const {
   223→    register,
   224→    handleSubmit,
   225→    watch,
   226→    reset,
   227→    formState: { errors },
   228→  } = useForm<TicketFormValues>({
   229→    resolver: zodResolver(ticketSchema),
   230→    defaultValues: {
   231→      subject: '',
   232→      body: '',
   233→      customerEmail: '',
   234→      customerName: '',
   235→    },
   236→  })
   237→
   238→  const [category, setCategory] = useState<string>('general')
   239→  const [priority, setPriority] = useState<string>('normal')
   240→  const [submitting, setSubmitting] = useState(false)
   241→  const [serverError, setServerError] = useState<string | null>(null)
   242→  const [fieldErrors, setFieldErrors] = useState<FieldErrors>({})
   243→  const [rateLimitReset, setRateLimitReset] = useState<number | null>(null)
   244→  const [success, setSuccess] = useState<SubmitResult | null>(null)
   245→
   246→  // Clear inline server field errors the moment the user edits that field.
   247→  const clearFieldError = (field: string) => {
   248→    if (fieldErrors[field]) {
   249→      setFieldErrors((prev) => {
   250→        const next = { ...prev }
   251→        delete next[field]
   252→        return next
   253→      })
   254→    }
   255→  }
   256→
   257→  const subjectLen = watch('subject')?.length ?? 0
   258→  const bodyLen = watch('body')?.length ?? 0
   259→
   260→  const onSubmit = async (values: TicketFormValues) => {
   261→    setServerError(null)
   262→    setFieldErrors({})
   263→    setSubmitting(true)
   264→    try {
   265→      const res = await fetch('/api/tickets', {
   266→        method: 'POST',
   267→        headers: { 'Content-Type': 'application/json' },
   268→        body: JSON.stringify({
   269→          ...values,
   270→          category,
   271→          priority,
   272→        }),
   273→      })
   274→      const payload: unknown = await res.json().catch(() => null)
   275→      if (res.ok && payload && typeof payload === 'object' && 'data' in payload) {
   276→        const data = (payload as { data: SubmitResult }).data
   277→        setSuccess(data)
   278→        reset()
   279→        setCategory('general')
   280→        setPriority('normal')
   281→        toast.success('Ticket submitted')
   282→        return
   283→      }
   284→      // 422 with field errors
   285→      if (res.status === 422 && payload && typeof payload === 'object' && 'fields' in payload) {
   286→        const fields = (payload as { fields: FieldErrors }).fields ?? {}
   287→        setFieldErrors(fields)
   288→        return
   289→      }
   290→      if (res.status === 429) {
   291→        const retryAfter = Number(res.headers.get('Retry-After') ?? '60')
   292→        setRateLimitReset(Math.max(1, retryAfter))
   293→        return
   294→      }
   295→      const msg =
   296→        payload && typeof payload === 'object' && 'error' in payload
   297→          ? String((payload as { error: unknown }).error)
   298→          : "We couldn't submit your ticket. Please try again."
   299→      setServerError(msg)
   300→    } catch {
   301→      setServerError('Network error. Please check your connection and try again.')
   302→    } finally {
   303→      setSubmitting(false)
   304→    }
   305→  }
   306→
   307→  // ---- Success view -------------------------------------------------
   308→  if (success) {
   309→    return (
   310→      <SubmitSuccessCard
   311→        result={success}
   312→        onReset={() => setSuccess(null)}
   313→        onView={() => {
   314→          setSuccess(null)
   315→          onSubmitted(success)
   316→        }}
   317→      />
   318→    )
   319→  }
   320→
   321→  // ---- Form view ----------------------------------------------------
   322→  const disabled = submitting || rateLimitReset !== null
   323→
   324→  return (
   325→    <Card className="border-border/60 shadow-sm">
   326→      <CardHeader>
   327→        <CardTitle className="flex items-center gap-2 text-lg">
   328→          <MessageSquarePlus className="h-5 w-5 text-emerald-600" />
   329→          Submit a ticket
   330→        </CardTitle>
   331→        <CardDescription>
   332→          Tell us what's going on. An agent will reply as soon as possible.
   333→        </CardDescription>
   334→      </CardHeader>
   335→      <CardContent>
   336→        <form
   337→          onSubmit={handleSubmit(onSubmit)}
   338→          className="grid gap-5"
   339→          noValidate
   340→        >
   341→          {/* Subject */}
   342→          <div className="grid gap-2">
   343→            <div className="flex items-baseline justify-between">
   344→              <Label htmlFor="subject">Subject *</Label>
   345→              <span className="text-[11px] text-muted-foreground tabular-nums">
   346→                {subjectLen}/{LIMITS.MAX_SUBJECT}
   347→              </span>
   348→            </div>
   349→            <Input
   350→              id="subject"
   351→              autoComplete="off"
   352→              maxLength={LIMITS.MAX_SUBJECT}
   353→              aria-invalid={!!errors.subject || !!fieldErrors.subject}
   354→              disabled={disabled}
   355→              {...register('subject', {
   356→                onChange: () => clearFieldError('subject'),
   357→              })}
   358→              placeholder="A short summary of your issue"
   359→            />
   360→            <FieldError msg={errors.subject?.message ?? fieldErrors.subject ?? null} />
   361→          </div>
   362→
   363→          {/* Body */}
   364→          <div className="grid gap-2">
   365→            <div className="flex items-baseline justify-between">
   366→              <Label htmlFor="body">Details *</Label>
   367→              <span className="text-[11px] text-muted-foreground tabular-nums">
   368→                {bodyLen}/{LIMITS.MAX_BODY}
   369→              </span>
   370→            </div>
   371→            <Textarea
   372→              id="body"
   373→              rows={6}
   374→              maxLength={LIMITS.MAX_BODY}
   375→              aria-invalid={!!errors.body || !!fieldErrors.body}
   376→              disabled={disabled}
   377→              {...register('body', {
   378→                onChange: () => clearFieldError('body'),
   379→              })}
   380→              placeholder="What were you trying to do? What happened instead? Any error messages you can paste here help."
   381→            />
   382→            <FieldError msg={errors.body?.message ?? fieldErrors.body ?? null} />
   383→          </div>
   384→
   385→          {/* Category + Priority */}
   386→          <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
   387→            <div className="grid gap-2">
   388→              <Label htmlFor="category">Category</Label>
   389→              <Select
   390→                value={category}
   391→                onValueChange={setCategory}
   392→                disabled={disabled}
   393→              >
   394→                <SelectTrigger id="category" className="w-full">
   395→                  <SelectValue placeholder="Pick a category" />
   396→                </SelectTrigger>
   397→                <SelectContent>
   398→                  <SelectItem value="general">General</SelectItem>
   399→                  <SelectItem value="billing">Billing</SelectItem>
   400→                  <SelectItem value="bug">Bug report</SelectItem>
   401→                  <SelectItem value="account">Account</SelectItem>
   402→                  <SelectItem value="feature">Feature request</SelectItem>
   403→                </SelectContent>
   404→              </Select>
   405→            </div>
   406→            <div className="grid gap-2">
   407→              <Label htmlFor="priority">Priority</Label>
   408→              <Select
   409→                value={priority}
   410→                onValueChange={setPriority}
   411→                disabled={disabled}
   412→              >
   413→                <SelectTrigger id="priority" className="w-full">
   414→                  <SelectValue placeholder="Pick a priority" />
   415→                </SelectTrigger>
   416→                <SelectContent>
   417→                  <SelectItem value="low">Low — minor inconvenience</SelectItem>
   418→                  <SelectItem value="normal">Normal — should work</SelectItem>
   419→                  <SelectItem value="high">High — blocking work</SelectItem>
   420→                  <SelectItem value="urgent">Urgent — outage / data loss</SelectItem>
   421→                </SelectContent>
   422→              </Select>
   423→            </div>
   424→          </div>
   425→
   426→          {/* Email + Name */}
   427→          <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
   428→            <div className="grid gap-2">
   429→              <Label htmlFor="customerEmail">Email *</Label>
   430→              <Input
   431→                id="customerEmail"
   432→                type="email"
   433→                autoComplete="email"
   434→                aria-invalid={!!errors.customerEmail || !!fieldErrors.customerEmail}
   435→                disabled={disabled}
   436→                {...register('customerEmail', {
   437→                  onChange: () => clearFieldError('customerEmail'),
   438→                })}
   439→                placeholder="you@example.com"
   440→              />
   441→              <FieldError msg={errors.customerEmail?.message ?? fieldErrors.customerEmail ?? null} />
   442→            </div>
   443→            <div className="grid gap-2">
   444→              <Label htmlFor="customerName">Name (optional)</Label>
   445→              <Input
   446→                id="customerName"
   447→                autoComplete="name"
   448→                disabled={disabled}
   449→                {...register('customerName')}
   450→                placeholder="What should we call you?"
   451→              />
   452→              <FieldError msg={errors.customerName?.message ?? null} />
   453→            </div>
   454→          </div>
   455→
   456→          {/* Footer / actions */}
   457→          {serverError && (
   458→            <Alert variant="destructive">
   459→              <AlertCircle className="h-4 w-4" />
   460→              <AlertTitle>Couldn&apos;t submit</AlertTitle>
   461→              <AlertDescription>{serverError}</AlertDescription>
   462→            </Alert>
   463→          )}
   464→          {rateLimitReset !== null && (
   465→            <Alert>
   466→              <TriangleAlert className="h-4 w-4 text-amber-600" />
   467→              <AlertTitle>Slow down a moment</AlertTitle>
   468→              <AlertDescription>
   469→                You&apos;ve submitted a few tickets in a row. Please wait{' '}
   470→                <RetryCountdown
   471→                  seconds={rateLimitReset}
   472→                  onDone={() => setRateLimitReset(null)}
   473→                />{' '}
   474→                before submitting another.
   475→              </AlertDescription>
   476→            </Alert>
   477→          )}
   478→          <div className="flex flex-col gap-2 sm:flex-row sm:items-center sm:justify-end">
   479→            <Button type="submit" disabled={disabled} className="sm:w-auto">
   480→              {submitting ? (
   481→                <>
   482→                  <Loader2 className="h-4 w-4 animate-spin" />
   483→                  Submitting…
   484→                </>
   485→              ) : (
   486→                <>
   487→                  Submit ticket
   488→                  <ArrowRight className="h-4 w-4" />
   489→                </>
   490→              )}
   491→            </Button>
   492→          </div>
   493→          <p className="text-[11px] text-muted-foreground">
   494→            By submitting, you agree we may keep a record of this conversation to
   495→            help you and improve our service. We don&apos;t share your details.
   496→          </p>
   497→        </form>
   498→      </CardContent>
   499→    </Card>
   500→  )
   501→}

   512→function SubmitSuccessCard({
   513→  result,
   514→  onReset,
   515→  onView,
   516→}: {
   517→  result: SubmitResult
   518→  onReset: () => void
   519→  onView: () => void
   520→}) {
   521→  const [copied, setCopied] = useState<'ref' | 'token' | null>(null)
   522→  const copy = async (which: 'ref' | 'token', value: string) => {
   523→    try {
   524→      await navigator.clipboard.writeText(value)
   525→      setCopied(which)
   526→      toast.success(which === 'ref' ? 'Reference copied' : 'Token copied')
   527→      setTimeout(() => setCopied(null), 1500)
   528→    } catch {
   529→      toast.error('Could not copy — please copy manually')
   530→    }
   531→  }
   532→
   533→  return (
   534→    <Card className="border-emerald-200 bg-emerald-50/40 dark:border-emerald-900 dark:bg-emerald-950/20">
   535→      <CardHeader>
   536→        <CardTitle className="flex items-center gap-2 text-lg">
   537→          <CheckCircle2 className="h-5 w-5 text-emerald-600" />
   538→          Ticket submitted
   539→        </CardTitle>
   540→        <CardDescription>
   541→          We&apos;ve logged your ticket. Save the details below so you can come
   542→          back to it.
   543→        </CardDescription>
   544→      </CardHeader>
   545→      <CardContent className="grid gap-5">
   546→        <div className="grid gap-2">
   547→          <Label className="text-xs text-muted-foreground">Ticket reference</Label>
   548→          <div className="flex items-center gap-2">
   549→            <code className="rounded-md border border-emerald-200 bg-background px-3 py-2 font-mono text-lg font-semibold tracking-wide dark:border-emerald-900">
   550→              {result.ref}
   551→            </code>
   552→            <Button
   553→              type="button"
   554→              variant="outline"
   555→              size="icon"
   556→              onClick={() => copy('ref', result.ref)}
   557→              aria-label="Copy ticket reference"
   558→            >
   559→              <ClipboardCopy className="h-4 w-4" />
   560→            </Button>
   561→            {copied === 'ref' && (
   562→              <CheckCircle2 className="h-4 w-4 text-emerald-600" />
   563→            )}
   564→          </div>
   565→        </div>
   566→
   567→        <div className="grid gap-2">
   568→          <Label className="text-xs text-muted-foreground">Your lookup token</Label>
   569→          <div className="flex items-start gap-2">
   570→            <code className="block w-full break-all rounded-md border border-emerald-200 bg-background px-3 py-2 font-mono text-xs dark:border-emerald-900">
   571→              {result.lookupToken}
   572→            </code>
   573→            <Button
   574→              type="button"
   575→              variant="outline"
   576→              size="icon"
   577→              onClick={() => copy('token', result.lookupToken)}
   578→              aria-label="Copy lookup token"
   579→              className="shrink-0"
   580→            >
   581→              <ClipboardCopy className="h-4 w-4" />
   582→            </Button>
   583→          </div>
   584→        </div>
   585→
   586→        <Alert>
   587→          <TriangleAlert className="h-4 w-4 text-amber-600" />
   588→          <AlertTitle>Save this token</AlertTitle>
   589→          <AlertDescription>
   590→            You&apos;ll need it (along with the reference) to check your ticket
   591→            status. We don&apos;t email it in this build, and we can&apos;t
   592→            recover it if you lose it.
   593→          </AlertDescription>
   594→        </Alert>
   595→
   596→        <div className="flex flex-col gap-2 sm:flex-row sm:justify-end">
   597→          <Button type="button" variant="ghost" onClick={onReset}>
   598→            Submit another
   599→          </Button>

   938→    const text = input.trim()

   943→    // Send only role+content to the API (UI-local id is dropped).
   944→    const apiMessages: ChatMessage[] = [
   945→      ...messages.map((m) => ({ role: m.role, content: m.content })),
   946→      { role: 'user', content: text },

   949→    setInput('')
   950→    setSending(true)

   952→    try {
   953→      const res = await fetch('/api/chat', {
   954→        method: 'POST',
   955→        headers: { 'Content-Type': 'application/json' },
   956→        body: JSON.stringify({ messages: apiMessages }),

   958→      const payload: unknown = await res.json().catch(() => null)
   959→      if (res.ok && payload && typeof payload === 'object' && 'data' in payload) {
   960→        const data = (payload as { data: ChatResult }).data
   961→        if (data.reply) {
   962→          setMessages((prev) => [
   963→            ...prev,
