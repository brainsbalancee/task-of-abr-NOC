     5→     5→ *
     6→     6→ * Three tabs:
     7→     7→ *   1. Submit a ticket   → POST /api/tickets

   145→   145→/**
   146→   146→ * Rate-limit retry countdown.
   147→   147→ *
   148→   148→ * Implemented as a component (not a hook) so the state initializer runs on
   149→   149→ * mount — no synchronous setState-in-effect, which keeps the linter happy
   150→   150→ * and avoids a cascading render. The parent conditionally renders this
   151→   151→ * (only when `rateLimitReset` is set), so each 429 mounts a fresh countdown
   152→   152→ * with the right starting value.
   153→   153→ */
   154→   154→function RetryCountdown({
   155→   155→  seconds,
   156→   156→  onDone,
   157→   157→}: {
   158→   158→  seconds: number
   159→   159→  onDone: () => void
   160→   160→}) {
   161→   161→  const [remaining, setRemaining] = useState(seconds)
   162→   162→  // Hold onDone in a ref so the interval effect can stay mount-only.
   163→   163→  const onDoneRef = useRef(onDone)
   164→   164→  useEffect(() => {
   165→   165→    onDoneRef.current = onDone
   166→   166→  }, [onDone])
   167→   167→
   168→   168→  useEffect(() => {
   169→   169→    if (seconds <= 0) {
   170→   170→      onDoneRef.current()
   171→   171→      return
   172→   172→    }
   173→   173→    const id = setInterval(() => {
   174→   174→      setRemaining((r) => {
   175→   175→        if (r <= 1) {
   176→   176→          clearInterval(id)
   177→   177→          onDoneRef.current()
   178→   178→          return 0
   179→   179→        }
   180→   180→        return r - 1
   181→   181→      })
   182→   182→    }, 1000)
   183→   183→    return () => clearInterval(id)
   184→   184→  }, [seconds])
   185→   185→
   186→   186→  return (
   187→   187→    <span className="font-semibold tabular-nums" aria-live="polite">
   188→   188→      {remaining}s
   189→   189→    </span>
   190→   190→  )
   191→   191→}

   197→   197→const ticketSchema = z.object({

   209→   209→    .email('That email address does not look right.'),
   210→   210→  customerName: z
   211→   211→    .string()
   212→   212→    .max(LIMITS.MAX_NAME, `Name must be under ${LIMITS.MAX_NAME} characters.`)
   213→   213→    .optional(),
   214→   214→})

   222→   222→function SubmitTicketForm({ onSubmitted }: SubmitTabProps) {
   223→   223→  const {
   224→   224→    register,
   225→   225→    handleSubmit,
   226→   226→    watch,
   227→   227→    reset,
   228→   228→    formState: { errors },
   229→   229→  } = useForm<TicketFormValues>({
   230→   230→    resolver: zodResolver(ticketSchema),
   231→   231→    defaultValues: {
   232→   232→      subject: '',
   233→   233→      body: '',
   234→   234→      customerEmail: '',
   235→   235→      customerName: '',
   236→   236→    },
   237→   237→  })
   238→   238→
   239→   239→  const [category, setCategory] = useState<string>('general')
   240→   240→  const [priority, setPriority] = useState<string>('normal')
   241→   241→  const [submitting, setSubmitting] = useState(false)
   242→   242→  const [serverError, setServerError] = useState<string | null>(null)
   243→   243→  const [fieldErrors, setFieldErrors] = useState<FieldErrors>({})
   244→   244→  const [rateLimitReset, setRateLimitReset] = useState<number | null>(null)
   245→   245→  const [success, setSuccess] = useState<SubmitResult | null>(null)
   246→   246→
   247→   247→  // Clear inline server field errors the moment the user edits that field.
   248→   248→  const clearFieldError = (field: string) => {
   249→   249→    if (fieldErrors[field]) {
   250→   250→      setFieldErrors((prev) => {
   251→   251→        const next = { ...prev }
   252→   252→        delete next[field]
   253→   253→        return next
   254→   254→      })
   255→   255→    }
   256→   256→  }
   257→   257→
   258→   258→  const subjectLen = watch('subject')?.length ?? 0
   259→   259→  const bodyLen = watch('body')?.length ?? 0
   260→   260→
   261→   261→  const onSubmit = async (values: TicketFormValues) => {
   262→   262→    setServerError(null)
   263→   263→    setFieldErrors({})
   264→   264→    setSubmitting(true)
   265→   265→    try {
   266→   266→      const res = await fetch('/api/tickets', {
   267→   267→        method: 'POST',
   268→   268→        headers: { 'Content-Type': 'application/json' },
   269→   269→        body: JSON.stringify({
   270→   270→          ...values,
   271→   271→          category,
   272→   272→          priority,
   273→   273→        }),
   274→   274→      })
   275→   275→      const payload: unknown = await res.json().catch(() => null)
   276→   276→      if (res.ok && payload && typeof payload === 'object' && 'data' in payload) {
   277→   277→        const data = (payload as { data: SubmitResult }).data
   278→   278→        setSuccess(data)
   279→   279→        reset()
   280→   280→        setCategory('general')
   281→   281→        setPriority('normal')
   282→   282→        toast.success('Ticket submitted')
   283→   283→        return
   284→   284→      }
   285→   285→      // 422 with field errors
   286→   286→      if (res.status === 422 && payload && typeof payload === 'object' && 'fields' in payload) {
   287→   287→        const fields = (payload as { fields: FieldErrors }).fields ?? {}
   288→   288→        setFieldErrors(fields)
   289→   289→        return
   290→   290→      }
   291→   291→      if (res.status === 429) {
   292→   292→        const retryAfter = Number(res.headers.get('Retry-After') ?? '60')
   293→   293→        setRateLimitReset(Math.max(1, retryAfter))
   294→   294→        return
   295→   295→      }
   296→   296→      const msg =
   297→   297→        payload && typeof payload === 'object' && 'error' in payload
   298→   298→          ? String((payload as { error: unknown }).error)
   299→   299→          : "We couldn't submit your ticket. Please try again."
   300→   300→      setServerError(msg)
   301→   301→    } catch {
   302→   302→      setServerError('Network error. Please check your connection and try again.')
   303→   303→    } finally {
   304→   304→      setSubmitting(false)
   305→   305→    }
   306→   306→  }
   307→   307→
   308→   308→  // ---- Success view -------------------------------------------------
   309→   309→  if (success) {
   310→   310→    return (
   311→   311→      <SubmitSuccessCard
   312→   312→        result={success}
   313→   313→        onReset={() => setSuccess(null)}
   314→   314→        onView={() => {
   315→   315→          setSuccess(null)
   316→   316→          onSubmitted(success)
   317→   317→        }}
   318→   318→      />
   319→   319→    )
   320→   320→  }
   321→   321→
   322→   322→  // ---- Form view ----------------------------------------------------
   323→   323→  const disabled = submitting || rateLimitReset !== null
   324→   324→
   325→   325→  return (
   326→   326→    <Card className="border-border/60 shadow-sm">
   327→   327→      <CardHeader>
   328→   328→        <CardTitle className="flex items-center gap-2 text-lg">
   329→   329→          <MessageSquarePlus className="h-5 w-5 text-emerald-600" />
   330→   330→          Submit a ticket
   331→   331→        </CardTitle>
   332→   332→        <CardDescription>
   333→   333→          Tell us what's going on. An agent will reply as soon as possible.
   334→   334→        </CardDescription>
   335→   335→      </CardHeader>
   336→   336→      <CardContent>
   337→   337→        <form
   338→   338→          onSubmit={handleSubmit(onSubmit)}
   339→   339→          className="grid gap-5"
   340→   340→          noValidate
   341→   341→        >
   342→   342→          {/* Subject */}
   343→   343→          <div className="grid gap-2">
   344→   344→            <div className="flex items-baseline justify-between">
   345→   345→              <Label htmlFor="subject">Subject *</Label>
   346→   346→              <span className="text-[11px] text-muted-foreground tabular-nums">
   347→   347→                {subjectLen}/{LIMITS.MAX_SUBJECT}
   348→   348→              </span>
   349→   349→            </div>
   350→   350→            <Input
   351→   351→              id="subject"
   352→   352→              autoComplete="off"
   353→   353→              maxLength={LIMITS.MAX_SUBJECT}
   354→   354→              aria-invalid={!!errors.subject || !!fieldErrors.subject}
   355→   355→              disabled={disabled}
   356→   356→              {...register('subject', {
   357→   357→                onChange: () => clearFieldError('subject'),
   358→   358→              })}
   359→   359→              placeholder="A short summary of your issue"
   360→   360→            />
   361→   361→            <FieldError msg={errors.subject?.message ?? fieldErrors.subject ?? null} />
   362→   362→          </div>
   363→   363→
   364→   364→          {/* Body */}
   365→   365→          <div className="grid gap-2">
   366→   366→            <div className="flex items-baseline justify-between">
   367→   367→              <Label htmlFor="body">Details *</Label>
   368→   368→              <span className="text-[11px] text-muted-foreground tabular-nums">
   369→   369→                {bodyLen}/{LIMITS.MAX_BODY}
   370→   370→              </span>
   371→   371→            </div>
   372→   372→            <Textarea
   373→   373→              id="body"
   374→   374→              rows={6}
   375→   375→              maxLength={LIMITS.MAX_BODY}
   376→   376→              aria-invalid={!!errors.body || !!fieldErrors.body}
   377→   377→              disabled={disabled}
   378→   378→              {...register('body', {
   379→   379→                onChange: () => clearFieldError('body'),
   380→   380→              })}
   381→   381→              placeholder="What were you trying to do? What happened instead? Any error messages you can paste here help."
   382→   382→            />
   383→   383→            <FieldError msg={errors.body?.message ?? fieldErrors.body ?? null} />
   384→   384→          </div>
   385→   385→
   386→   386→          {/* Category + Priority */}
   387→   387→          <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
   388→   388→            <div className="grid gap-2">
   389→   389→              <Label htmlFor="category">Category</Label>
   390→   390→              <Select

   396→   396→                  <SelectValue placeholder="Pick a category" />
   397→   397→                </SelectTrigger>
   398→   398→                <SelectContent>

   938→   938→/**
   939→   939→ * Customer reply composer. Mirrors the staff reply box pattern:
   940→   940→ *  - Textarea with character counter (`LIMITS.MAX_BODY`).
   941→   941→ *  - "Send reply" button POSTs to /api/tickets/reply with `{ref, lookupToken, body}`.
   942→   942→ *  - 201 → optimistic append to the thread, clear the box, success toast.
   943→   943→ *  - 429 → RetryCountdown (same component the rest of the surface uses).
   944→   944→ *  - 409 → inline "ticket is closed" notice + hide the composer.
   945→   945→ *  - 422 → inline field error next to the textarea.
   946→   946→ *  - 5xx / network → toast error, keep the typed text so the user can retry.
   947→   947→ *
   948→   948→ * Note: the API may flip the ticket status from 'resolved' → 'open' on a
   949→   949→ * successful reply (a customer follow-up on a "resolved" ticket reopens
   950→   950→ * it). The parent's `onSent` callback handles that state update.
   951→   951→ */
   952→   952→function ReplyComposer({
   953→   953→  ref_,
   954→   954→  lookupToken,
   955→   955→  onSent,
   956→   956→}: {
   957→   957→  ref_: string
   958→   958→  lookupToken: string
   959→   959→  onSent: (msg: TicketMessageView) => void
   960→   960→}) {
   961→   961→  const [body, setBody] = useState('')
   962→   962→  const [sending, setSending] = useState(false)
   963→   963→  const [fieldError, setFieldError] = useState<string | null>(null)
   964→   964→  const [rateLimitReset, setRateLimitReset] = useState<number | null>(null)
   965→   965→  const [closed, setClosed] = useState(false)
   966→   966→
   967→   967→  const onSend = async () => {
   968→   968→    const trimmed = body.trim()
   969→   969→    if (!trimmed || sending || rateLimitReset !== null) return
   970→   970→    setSending(true)
   971→   971→    setFieldError(null)
   972→   972→    try {
   973→   973→      const res = await fetch('/api/tickets/reply', {
   974→   974→        method: 'POST',
   975→   975→        headers: { 'Content-Type': 'application/json' },
   976→   976→        body: JSON.stringify({
   977→   977→          ref: ref_,
   978→   978→          lookupToken,
   979→   979→          body: trimmed,
